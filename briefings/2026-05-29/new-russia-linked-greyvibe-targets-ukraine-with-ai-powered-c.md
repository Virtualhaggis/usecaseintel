```spl

```

**Defender KQL:**
```kql
DeviceProcessEvents
| where Timestamp > ago(7d)
| where InitiatingProcessFileName =~ "explorer.exe"
| where FileName in~ ("powershell.exe","pwsh.exe","mshta.exe","wscript.exe","cscript.exe")
| where ProcessCommandLine has_any ("iex ","Invoke-Expression","DownloadString","FromBase64String","-EncodedCommand","-enc ","-W Hidden","-WindowStyle Hidden","curl ","Invoke-RestMethod")
| join kind=leftouter (DeviceNetworkEvents | where Timestamp > ago(7d) | where RemoteUrl has_any ("zoomconference.click","zoomconference.app","lapas.live") | project DeviceId, ClickTime=Timestamp, BadUrl=RemoteUrl) on DeviceId
| where isnull(ClickTime) or Timestamp between (ClickTime .. ClickTime + 5m)

```
