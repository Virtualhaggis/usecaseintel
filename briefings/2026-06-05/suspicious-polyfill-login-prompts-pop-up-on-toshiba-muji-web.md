## Recommended hunts

### [LLM] Endpoint egress to reactivated polyfill[.]io hijacked CDN domain

`UC_11_2` · phase: **delivery** · confidence: **High**

**Splunk SPL (CIM):**
```spl
| tstats summariesonly=true count min(_time) as firstTime max(_time) as lastTime values(Web.url) as urls values(Web.http_user_agent) as agents values(Web.status) as statuses from datamodel=Web where (Web.dest IN ("polyfill.io","cdn.polyfill.io") OR Web.url="*://polyfill.io/*" OR Web.url="*://cdn.polyfill.io/*") by Web.src Web.user Web.dest host
| `drop_dm_object_name(Web)`
| eval firstTime=strftime(firstTime,"%Y-%m-%d %H:%M:%S"), lastTime=strftime(lastTime,"%Y-%m-%d %H:%M:%S")
| where lastTime > "2026-05-20 00:00:00"
| sort - count
```

**Defender KQL:**
```kql
let HijackedDomains = dynamic(["polyfill.io","cdn.polyfill.io"]);
DeviceNetworkEvents
| where Timestamp > ago(7d)
| where RemoteUrl has_any (HijackedDomains)
   or RemoteUrl matches regex @"(?i)(^|//|\.)polyfill\.io(/|$|:)"
| where not(RemoteUrl has_any ("polyfill.com","polyfill.top"))
| project Timestamp, DeviceName, InitiatingProcessFileName, InitiatingProcessAccountName,
          InitiatingProcessCommandLine, InitiatingProcessParentFileName,
          RemoteUrl, RemoteIP, RemotePort, ActionType
| order by Timestamp desc
```

### [LLM] HTTP 401 Basic-Auth challenge served by polyfill[.]io to first-party origin

`UC_11_3` · phase: **actions** · confidence: **High**

**Splunk SPL (CIM):**
```spl
| tstats summariesonly=true count min(_time) as firstTime max(_time) as lastTime values(Web.http_referrer) as referrers values(Web.user) as users values(Web.http_user_agent) as agents from datamodel=Web where (Web.dest IN ("polyfill.io","cdn.polyfill.io") OR Web.url="*://polyfill.io/*" OR Web.url="*://cdn.polyfill.io/*") AND Web.status=401 by Web.src Web.dest Web.url host
| `drop_dm_object_name(Web)`
| eval firstTime=strftime(firstTime,"%Y-%m-%d %H:%M:%S"), lastTime=strftime(lastTime,"%Y-%m-%d %H:%M:%S")
| sort - count
```

**Defender KQL:**
```kql
// Note: Defender DeviceNetworkEvents does not surface HTTP status. We surrogate-detect by
// pivoting on browser-process loads of polyfill.io with NO subsequent same-host file/script write,
// which is the network-layer fingerprint of a 401 with empty body (the prompt-only response).
let HijackedDomains = dynamic(["polyfill.io","cdn.polyfill.io"]);
let PolyfillFetches = DeviceNetworkEvents
    | where Timestamp > ago(7d)
    | where RemoteUrl has_any (HijackedDomains)
    | where InitiatingProcessFileName in~ ("chrome.exe","msedge.exe","firefox.exe","brave.exe","iexplore.exe","opera.exe")
    | project FetchTime=Timestamp, DeviceId, DeviceName, InitiatingProcessId, InitiatingProcessFileName, InitiatingProcessAccountName, RemoteUrl, RemoteIP;
PolyfillFetches
| join kind=leftouter (
    DeviceFileEvents
    | where Timestamp > ago(7d)
    | where ActionType in ("FileCreated","FileModified")
    | where FileName endswith ".js" or FolderPath has "Cache"
    | project FollowupFileTime=Timestamp, DeviceId, InitiatingProcessId, FileName
) on DeviceId, InitiatingProcessId
| where isempty(FollowupFileTime) or datetime_diff('second', FollowupFileTime, FetchTime) > 30
| summarize FetchCount=count(), arg_max(FetchTime, *) by DeviceId, DeviceName, InitiatingProcessAccountName, RemoteUrl
| order by FetchTime desc
```

### [LLM] First-party web asset referencing abandoned polyfill[.]io CDN script tag (inventory hunt)

`UC_11_4` · phase: **weapon** · confidence: **Medium**

**Splunk SPL (CIM):**
```spl
| tstats summariesonly=true count min(_time) as firstTime max(_time) as lastTime values(Web.src) as clients dc(Web.src) as uniqueClients from datamodel=Web where (Web.dest IN ("polyfill.io","cdn.polyfill.io") OR Web.url="*polyfill.io/*") AND isnotnull(Web.http_referrer) AND Web.http_referrer!="-" AND Web.http_referrer!="" by Web.http_referrer Web.url
| `drop_dm_object_name(Web)`
| eval ReferrerDomain=replace(http_referrer,"^https?://([^/]+)/.*","\1")
| stats sum(count) as Hits dc(http_referrer) as DistinctReferrers values(url) as PolyfillUrls by ReferrerDomain
| where Hits > 0
| sort - Hits
```

**Defender KQL:**
```kql
// Inventory hunt — surface every browser process that fetched polyfill.io and the originating tab/page where possible.
// AdditionalFields on DeviceNetworkEvents sometimes carries Referer in Chromium-based browsers; otherwise pivot to the
// process command-line / parent for context.
let HijackedDomains = dynamic(["polyfill.io","cdn.polyfill.io"]);
DeviceNetworkEvents
| where Timestamp > ago(30d)
| where RemoteUrl has_any (HijackedDomains)
| where InitiatingProcessFileName in~ ("chrome.exe","msedge.exe","firefox.exe","brave.exe","iexplore.exe","opera.exe","node.exe","curl.exe","wget.exe","powershell.exe","python.exe")
| extend ParsedReferer = tostring(parse_json(AdditionalFields).http_referrer)
| summarize FirstSeen=min(Timestamp), LastSeen=max(Timestamp), HitCount=count(),
            UniqueDevices=dcount(DeviceId),
            SampleReferer=any(ParsedReferer),
            DeviceList=make_set(DeviceName, 20)
            by InitiatingProcessFileName, RemoteUrl
| order by HitCount desc
```
